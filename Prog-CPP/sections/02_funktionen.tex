\section{Funktionen}
\subsection{Default Argumente}
\begin{itemize}
    \item \textbf{Prototyp}: Standardwerte werden \textbf{nur} im Interface zugewiesen.
    \item \textbf{Weglassen}: Default-Parameter können beim Aufruf übersprungen werden.
    \item \textbf{Reihenfolge}: Default-Parameter müssen \textbf{\cor{immer am Ende (rechts)}} der Parameterliste stehen.
\end{itemize}

\subsection{Funzioni Inline}
\begin{itemize}
    \item \textbf{Makro-Ersatz}: Sichere Alternative zu C-Makros.
    \item \textbf{Zero Overhead}: Code wird direkt eingefügt (keine Funktionssprünge).
    \item \textbf{Type Safety}: Der Compiler garantiert die Typüberprüfung.
    \item \textbf{Ziel}: Kleine, sehr häufig aufgerufene Funktionen (z.B. Loops).
    \item \textbf{No-go}: Unanwendbar bei Rekursion oder Funktionspointern.
    \item \textbf{Header-only}: Wenn in Headern genutzt, Fkt.-Def. direkt ins file.h
    \item \textbf{Optimierung}: Erfordert aktive Flags (z.B. \texttt{-O3}).
\end{itemize}
