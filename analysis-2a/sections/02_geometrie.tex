    \begin{itemize}
        \item Elimination Parameter $t$ (kann Schwierig sein)
        \item \textbf{Eindeutige} Funktionsgleichung $y = f(x)$

    \begin{itemize}
        \item $F(x, y) = 0$
        \item nicht eindeutige Gleichung in $x$ und $y$ \\
            z.B. $x^2 + y^2 = 1$
    \end{itemize}
\end{minipage}

\begin{minipage}[t]{0.48\columnwidth}
    \subsubsection*{Parameterdarstellung}

    \begin{itemize}
        \item Darstellung als Vektor: $\begin{pmatrix} x(t) \\ y(t) \\ \dots \end{pmatrix}$
        \item $\vec{c}$ ableitbar nach $t$ (keine Knicke)
        \item $\vec{c}$ ist regulär, wenn ableitbar und $\vec{c'} \neq \vec{0}$ (nie $v=0$)
    \end{itemize}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\columnwidth}
    \subsubsection*{Polarform}

    \begin{itemize}
        \item Darstellung: Radius als Funktion des Winkels:  $r = f(\varphi)$
        \item Darstellung als Polar-Vektor: $\vec{c}(\varphi) = \begin{pmatrix} r(\varphi) \, \cos(\varphi) \\ r(\varphi) \, \sin(\varphi) \end{pmatrix}$
        \item Im Pol bedeutet $r = 0$ ; Winkel $\varphi_0$ z.B. $\pi$
        \item Drehung um Winkel $\alpha$: $r(\varphi) \rightarrow r(\varphi - \alpha)$ \\
            (Drehung  im Gegenuhrzeigersinn)
    \end{itemize}
\end{minipage}

\medskip

\begin{minipage}[t]{0.48\columnwidth}
    \subsubsection*{Tangentialvektor}

    \begin{itemize}
        \item Momentane Geschwindigkeit
        \item Darstellung als Vektor: $\frac{\diff}{\diff t}\vec{c} =  \begin{pmatrix} \dot{x}(t) \\ \dot{y}(t) \\ \dots \end{pmatrix}$
        \item Punkt steht für Ableitung nach der Zeit
    \end{itemize}

\subsubsection{Definitionsbereich $\varphi$}
\begin{enumerate}
    \item \textbf{Algebraische Existenz}: Nenner $\neq 0$ und Wurzelargumente $\ge 0$.
    \item \textbf{Positivität des Radius}: Da $r$ eine Distanz ist, setze $r(\varphi) \ge 0$.
    \[ f(\varphi) \ge 0 \qquad \text{(Zähler und Nenner mit demselben Vorzeichen)}\]
    \item \textbf{Bedingung Arkustangens}: Wenn $\varphi = \arctan(y/x)$, dann: \quad $\varphi \in \left( -\frac{\pi}{2}, \frac{\pi}{2} \right)$
    \item \textbf{Finaler Bereich}: Schnittmenge aller obigen Bedingungen.
    \[ \mathcal{D}_{\varphi} = \{ \text{Punkt 1} \cap \text{Punkt 2} \cap \text{Punkt 3} \} \]
\end{enumerate}

\subsection{Kurvenordnung}
Zwei Kurven $f(x)$ und $g(x)$ haben einen Kontakt der \textbf{Ordnung $n$} in einem Punkt $x_0$, wenn alle ihre Ableitungen bis zur Ordnung $n$ dort übereinstimmen, aber die $(n+1)$-te Ableitung verschieden ist.

\begin{equation*}
    f^{(k)}(x_0) = g^{(k)}(x_0) \quad \text{für alle } k = 0, 1, \dots, n \qquad f^{(n+1)}(x_0) \neq g^{(n+1)}(x_0)
\end{equation*}

\subsubsection{Schnelle Berechnungsmethoden}
\begin{itemize}
    \item \textbf{Vielfachheit der Nullstellen:} Hat die Gleichung $f(x) - g(x) = 0$ in $x_0$ eine Lösung mit Vielfachheit $k$, so ist die Kontaktordnung $n = k - 1$.
    \item \textbf{Taylor-Entwicklung:} Ist $g(x)$ das Taylor-Polynom vom Grad $n$ der Funktion $f(x)$ um $x_0$, so ist der Kontakt zwischen Funktion und Polynom (mindestens) von der Ordnung $n$.
\end{itemize}
