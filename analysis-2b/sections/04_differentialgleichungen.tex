\subsection{Picard-Lindelöf (Eindeutigkeit der Lösung)}

Schema zur Beurteilung der \textbf{Lösbarkeit} einer DGL an einem Punkt $(x_0, y_0)$:

\medskip
\textbf{Schritt 1: Stetigkeit (Existenz)}
\begin{itemize}
    \item[(1)] $f(x, y, y', \, ... \, y^{(n-1)} )$ lokal bei $x_0 ; y_0$ stetig.
    \item[] Stetigkeit rimane con operazioni elementari: $+, -, \cdot, :, \text{subst (\crd{Att}. }\mathbb{D}) (\sqrt{f}, \ln(f))$\\
    \cor{Lineare oder elementare Funktionen sind fast immer stetig.}
\end{itemize}

\textbf{Schritt 2: Beschränktheit (Eindeutigkeit)}
\begin{itemize}
    \item[(2)] Alle partiellen Ableitungen bilden: $f_y = \frac{\partial f}{\partial y}$ ; $f_{y'} = \frac{\partial f}{\partial y'}$ ; ...
    \begin{itemize}
        \item Ist das Ergebnis \textbf{konstant} \textrightarrow\ automatisch lokal beschränkt.
        \item Ist es eine \textbf{Formel} \textrightarrow\ $x_0, y_0$ einsetzen. 
        Das Ergebnis muss $\in \mathbb{R}$
    \end{itemize}
\end{itemize}

\medskip
\textbf{Fazit:}
\begin{itemize}
    \item \textbf{Wenn} (1) erfüllt \textbf{und} (2) lokal bei $x_0 ; y_0$ beschränkt: \\
    \textrightarrow\ Die DGL ist \textbf{lokal eindeutig lösbar} (inkl. Anfangsbedingungen).
    \item \textbf{Wenn} (1) erfüllt, \textbf{aber} (2) \myul{nicht} erfüllt (z.B. $\frac{1}{0}$ in der Ableitung): \\
    \textrightarrow\ Eindeutigkeit ist nicht garantiert. \textbf{Singularitäts-Test durchführen!}
\end{itemize}

\medskip
\subsubsection{Singularitäts-Test (Ablauf bei Polstelle in $f_y$):}
\begin{enumerate}
    \item \textbf{Polstelle finden:} Nenner der partiellen Ableitung $= 0$ setzen.
    \item \textbf{Auflösen:} Diese Gleichung nach $y$ auflösen (das liefert die singuläre Kandidaten-Lösung).
    \item \textbf{Verifizieren:} Diese Funktion $y$ in die \textit{ursprüngliche} DGL einsetzen und kontrollieren, ob die Gleichung erfüllt ist.
\end{enumerate}

\subsubsection{Trennung der Variablen (separiert)}
\label{Trennung der Variablen (separiert)}

\vspace{-0.2cm}

$$ y' = f(x \, ; \, y) = ( \text{Faktor in } x) * ( \text{Faktor in } y) = \tilde{f}(x) * g(y) $$ 

\renewcommand{\arraystretch}{1.7}
\begin{tabular}{ll}
    (1) & $\frac{y'}{g(y)} = \tilde{f}(x)$ mit $g(y) \neq 0$                                                                            \\

    (3) & $\int \limits_{y_0}^{y} \frac{\diff \tilde{y}}{g(y)} =$ gelöstes Integral                                                     \\
    (4) & Integral links lösen = gelöstes Integral rechts                                                                               \\
    (5) & Gleichung auflösen nach $y$ \textrightarrow\ Formel in $x$                                                                    \\
    (6) & Singularitätstest um weitere Lösungen zu finden (vedi \ref{chap:sing_erg})                                                                              \\
\end{tabular}

$$ y' = f(x \, ; \, y) = f \underbrace{\left(\frac{y}{x}\right)}_z \qquad \qquad y(x_0) = y_0 ; (x ; x_0 \neq 0) $$\vspace{-0.5em}

\begin{itemize}
    \item weiter in Abschnitt \ref{Trennung der Variablen (separiert)} mit $\tilde{f}(x) = \frac{1}{x}$, Substitution $y = z$

\para{Umkehrfunktions-Trick}

Variablen $x$ und $y$ vertauschen wenn:

\smallskip

\begin{itemize}
    \item $f(\frac{y}{x})$ verkehrt herum, also $f(\frac{x}{y})$
    \item Ableitung (Steigung) $y'$ steht im Kehrwert $\frac{1}{y'} \, \longrightarrow \, \tilde{x}' = \frac{1}{\tilde{y}'}$
\end{itemize}
